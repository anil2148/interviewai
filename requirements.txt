# =============================================================================

faster-whisper>=1.0.0

# numpy is a direct dependency of faster-whisper
numpy>=1.26.0

# SpeechRecognition used for Google STT fallback path only
# The library itself installs fine headlessly; audio capture is never called
# on the server because PTT/auto-listen are client-side in web-service mode.
SpeechRecognition>=3.10.0

# ── DO NOT include pyaudio or sounddevice on Render ──────────────────────────
# pyaudio requires PortAudio system headers (portaudio.h) which are not present
# in Render's Linux containers. sounddevice has the same dependency.
# Audio capture for PTT/auto-listen is handled client-side (browser MediaStream
# API) and sent to the server as a WAV blob via the /transcribe endpoint.
# If you need server-side audio capture on a custom Linux box:
#   sudo apt-get install portaudio19-dev libsndfile1
#   pip install pyaudio sounddevice
